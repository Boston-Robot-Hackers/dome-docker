# Each section: [name], then key_url, key_file, key_dearmor, list, packages fields.
# key_dearmor = yes: pipe key through gpg --dearmor (ASCII-armored keys).
# key_dearmor = no:  download key directly (already binary GPG format).
# {arch} expands to $(dpkg --print-architecture) at install time.
# {key_file} expands to the key_file value.
